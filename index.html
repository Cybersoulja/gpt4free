Quick Start

# 1. Clone the repository
git clone https://github.com/cybersoulja/gpt4free
cd gpt4free

# 2. Install dependencies
pip install -r requirements.txt

# 3. Use a provider (example: Phind)
import phind

phind.cf_clearance = 'your_cf_clearance'
phind.user_agent   = 'Mozilla/5.0 ...'

result = phind.Completion.create(
    model   = 'gpt-4',
    prompt  = 'Explain async/await in Python',
    results = phind.Search.create('Explain async/await in Python', actualSearch=True),
)

print(result.completion.choices[0].text)
